  }
  const from = (nom: string) => {
    let op: 'select' | 'insert' | 'update' | 'delete' = 'select'
    let charge: Ligne | Ligne[] | null = null
    let colonnes: string | null = null
    const filtres: ((l: Ligne) => boolean)[] = []
    const projeter = (l: Ligne): Ligne => {
      if (colonnes === null || colonnes.trim() === '*') return structuredClone(l)
      const o: Ligne = {}
      for (const c of colonnes.split(',').map((x) => x.trim())) o[c] = structuredClone(l[c] ?? null)
      return o

      const t = table(nom)
      if (op === 'insert') {
        const lignes = (Array.isArray(charge) ? charge : [charge!]).map((c) => ({
          id: randomUUID(),

          cree_le: new Date().toISOString(),
          ...structuredClone(c),
        }))
        t.push(...lignes)
        return { data: lignes.map(projeter), error: null }
      }
      const touchees = t.filter((l) => filtres.every((f) => f(l)))
      if (op === 'update') {
        for (const l of touchees) Object.assign(l, structuredClone(charge as Ligne))
        return { data: touchees.map(projeter), error: null }
      }
      if (op === 'delete') {
        base.tables.set(nom, t.filter((l) => !touchees.includes(l)))
        return { data: [], error: null }
      }
      return { data: touchees.map(projeter), error: null }
    }

      select(c?: string) { colonnes = c ?? '*'; return b },
      insert(c: Ligne | Ligne[]) { op = 'insert'; charge = c; return b },
      update(c: Ligne) { op = 'update'; charge = c; return b },
      delete() { op = 'delete'; return b },
      eq(k: string, v: unknown) { filtres.push((l) => l[k] === v); return b },
      neq(k: string, v: unknown) { filtres.push((l) => l[k] !== v); return b },
      in(k: string, v: unknown[]) { filtres.push((l) => v.includes(l[k])); return b },

      maybeSingle() { const r = executer(); return Promise.resolve({ data: r.data[0] ?? null, error: null }) },

const resume = (v: unknown): unknown => {
  if (v === null || v === undefined) return v ?? null
  if (typeof v === 'string') return v.length > 100 ? `${v.slice(0, 100)}… (${String(v.length)} car.)` : v
  if (Array.isArray(v)) return v.map(resume)
  if (typeof v === 'object') return Object.fromEntries(Object.entries(v as Ligne).map(([k, x]) => [k, resume(x)]))
  return v
}
const brute = (id: string): Ligne | null => {
  const l = (base.tables.get(TABLE_GENERATIONS) ?? []).find((x) => x.id === id)
  return l === undefined ? null : structuredClone(l)
}
const versions = (id: string) => (base.tables.get(TABLE_VERSIONS) ?? [])
  .filter((v) => v.generation_id === id)
  .map((v) => ({ rang: v.rang, etape: v.etape, cout_usd: v.cout_usd, cles_etat: Object.keys((v.etat ?? {}) as object) }))
const comparer = (a: Ligne | null, b: Ligne | null) => {
  const cles = [...new Set([...Object.keys(a ?? {}), ...Object.keys(b ?? {})])].sort()
  const inchangees: string[] = []
  const changees: Record<string, { avant: unknown; apres: unknown }> = {}
  for (const k of cles) {
    if (JSON.stringify(a?.[k]) === JSON.stringify(b?.[k])) inchangees.push(k)
    else changees[k] = { avant: resume(a?.[k]), apres: resume(b?.[k]) }
  }
  return { inchangees, changees }
}

const reponseDe = (s: 'ok' | 'A' | 'B') =>

describe('OBSERVATION — ce qu’une régénération écrase après la réparation', () => {
  const suites: Record<string, ('ok' | 'A' | 'B')[]> = {
    S1_regeneration_reussie: ['ok'],
    S2_regeneration_non_convergee: ['A'],
    S3_echec_puis_reussite: ['A', 'ok'],
    S4_deux_reussites: ['ok', 'ok'],
    S5_erreur_fournisseur: ['B'],
  }
  for (const [nom, suite] of Object.entries(suites)) {
    it(nom, async () => {
      reseau.reponses = reponseDe('ok')
      const d0 = (await (await generer()).json()) as { ok?: boolean; id?: string }
      expect(d0.ok).toBe(true)
      const id = d0.id!
      const etapes: unknown[] = []
      let precedente = brute(id)
      etapes.push({ etape: 'generation_initiale', ligne: resume(precedente), tours: toursDe(precedente).length, versions: versions(id) })
      for (const [i, s] of suite.entries()) {
        reseau.i = 0
        reseau.appels = 0
        reseau.reponses = reponseDe(s)
        const r = await generer(id)
        const corps = (await r.clone().json()) as Ligne
        const apres = brute(id)
        etapes.push({
          etape: `regeneration_${String(i + 1)}_${s}`,
          corps_rendu: { ok: corps.ok, error: resume(corps.error), coutUsd: corps.coutUsd },
          comparaison_avec_l_etat_precedent: comparer(precedente, apres),
          tours_dans_la_ligne: toursDe(apres).length,
          versions: versions(id),
          lignes_app_generations: (base.tables.get(TABLE_GENERATIONS) ?? []).map((x) => ({ id: x.id === id ? 'ID_OEUVRE' : 'AUTRE', statut: x.statut, ok: x.ok })),
        })
        precedente = apres
      }
      releves[nom] = etapes
      writeFileSync(SORTIE, JSON.stringify(releves, null, 2))
    }, 120_000)
  }
})
